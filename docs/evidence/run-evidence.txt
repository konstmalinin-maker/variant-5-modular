from pathlib import Path
import os,subprocess,sys,tempfile
root=Path('/workspace/variant-5-modular')
mode=sys.argv[1]
def run(args,env=None):
 print('$ '+' '.join(args),flush=True)
 result=subprocess.run(args,cwd=root,env=env,text=True,capture_output=True)
 print(result.stdout+result.stderr,end='',flush=True)
 print('exit code:',result.returncode,flush=True)
 return result
if mode=='layers':
 with tempfile.TemporaryDirectory(prefix='pr7-layer-') as folder:
  tmp=Path(folder);f=tmp/'src/domain/violation.py';f.parent.mkdir(parents=True)
  args=[sys.executable,'scripts/check_layers.py','--root',str(tmp)]
  print('A. Проверка корректной структуры',flush=True);assert run(args).returncode==0
  f.write_text('import sqlite3 as sqlite3\n')
  print('A. Добавлен прямой импорт sqlite3 в domain:',f,flush=True)
  assert run(args).returncode==1
  f.unlink();print('A. Запрещённый импорт удалён',flush=True);assert run(args).returncode==0
elif mode=='storage':
 for backend in (sys.argv[2:] or ['sqlite','json']):
  with tempfile.TemporaryDirectory(prefix='pr7-'+backend+'-') as folder:
   env={**os.environ,'STORAGE':backend,'DATA_PATH':folder+'/university.data'}
   print('\nG. STORAGE='+backend+' DATA_PATH='+env['DATA_PATH'],flush=True)
   cmd=[sys.executable,'main.py']
   student=run(cmd+['add-student','Иван'],env).stdout.strip()
   course=run(cmd+['add-course','Математика','3'],env).stdout.strip()
   assert run(cmd+['enroll',student,course],env).returncode==0
   assert run(cmd+['grade',student,course,'5'],env).returncode==0
   result=run(cmd+['gpa',student],env);assert '5.00' in result.stdout
   assert run(cmd+['list',student],env).returncode==0
   assert run(cmd+['export',student,folder+'/plan.csv'],env).returncode==0
   assert run(cmd+['drop',student,course],env).returncode==0
